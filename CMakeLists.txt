cmake_minimum_required(VERSION 3.16)
project(RUNESort LANGUAGES CXX)
include(CTest)
add_subdirectory(runesort_v5)

# Stable current API; legacy rune::sort in runesort_v5 remains compatible.
add_library(runesort_current INTERFACE)
add_library(runesort::current ALIAS runesort_current)
target_include_directories(runesort_current INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(runesort_current INTERFACE cxx_std_17)
if(BUILD_TESTING)
  foreach(variant native scalar portable)
    add_executable(test_current_${variant} tests/test_current.cpp)
    target_link_libraries(test_current_${variant} PRIVATE runesort::current)
    add_test(NAME current_${variant} COMMAND test_current_${variant})
  endforeach()
  target_compile_definitions(test_current_scalar PRIVATE RUNE27_DISABLE_SIMD)
  target_compile_definitions(test_current_portable PRIVATE RUNE27_DISABLE_BMI2 RUNE27_DISABLE_SIMD)
endif()
